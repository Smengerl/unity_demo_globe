// ARCHIV: prozedurale Wolken aus dem Earth-Shader (2026-10-25), entfernt weil zu teuer fuer Handys
// (grob 38 Rauschfunktionen + 4x Wirbelschleife pro Pixel). Rekonstruiert aus der letzten funktionierenden Fassung,
// nicht in dieser Form neu kompiliert. Gedacht als Vorlage, um daraus OFFLINE eine Wolkentextur zu backen
// (equirektangular, 4096x2048, R8) und im Spiel nur noch zwei driftende Texturschichten zu sampeln.
//
// Benoetigte Globals (von SkyDriver gesetzt, ebenfalls entfernt):
//   float _TycoonCloudFlow;   // rad Winddrift = real*0.004 + simSekunden*5.4e-6 (unbegrenzt)
//   float _TycoonCloudMorph;  // real*2 + simSekunden*0.002 (treibt langsame Formaenderung)
// Benoetigt TyHash/TyNoise aus TycoonCommon.hlsl (Value-Noise, 3D).
// Uniform:  float _CloudAmount (0..1, Standard 0.6)
//
// Idee in Kuerze:
//  - Deckung nach Breitengrad (Tropen/Sturmzonen viel, Subtropen wenig)
//  - 14 Zyklone in einem "statischen Muster": die Abtastrichtung wird um jedes Zentrum verdrillt -> Spiralarme
//  - zwei Rauschfelder: grosse Wettersysteme (syn) + feine, stark gewarpte Schleier (fine); Tropen klumpiger
//  - Windrichtung je Breitengrad (Passat West, Westwind Ost) per Flow-Map-Technik: zwei um eine halbe Periode
//    versetzte Kopien, ueberblendet, damit die Scherung begrenzt bleibt
//  - jede Periode ("weather") hat eigenen Rauschversatz und andere Zyklon-Positionen -> nie dasselbe Muster
//  - Wolkenschatten: dieselbe Dichte, vom Sonnenstand weg versetzt, abgedunkelt
//  - Darstellung: Mischung zu Weiss mit max. 50 % Deckkraft (pow(cloud,1.3)*0.5), damit das Land durchscheint

float CloudCoverageLat(float lat)   // lat in rad
{
    float a = abs(lat);
    float c = 0.5;
    c += 0.30 * exp(-pow(a / 0.15, 2.0));              // ITCZ
    c += 0.35 * exp(-pow((a - 0.87) / 0.22, 2.0));     // Sturmzonen ~50 Grad
    c -= 0.35 * exp(-pow((a - 0.43) / 0.14, 2.0));     // subtropische Trockenguertel ~25 Grad
    return c;
}

static const float4 kVortex[14] =   // x=Breite Grad, y=Laenge Grad, z=Radius rad, w=Verdrillung im Zentrum rad (+Nord/-Sued)
{
    float4( 52.0,  -40.0, 0.24,  2.6), float4( 58.0,   10.0, 0.20,  2.4), float4( 47.0,   70.0, 0.22,  2.6),
    float4( 55.0,  130.0, 0.24,  2.4), float4( 50.0, -110.0, 0.22,  2.6), float4( 62.0, -160.0, 0.19,  2.2),
    float4(-50.0,  -30.0, 0.24, -2.6), float4(-55.0,   40.0, 0.22, -2.6), float4(-48.0,  110.0, 0.24, -2.4),
    float4(-58.0,  170.0, 0.20, -2.4), float4(-52.0, -100.0, 0.22, -2.6),
    float4( 18.0,  -60.0, 0.08,  3.6), float4(-15.0,   80.0, 0.08, -3.6), float4( 15.0,  130.0, 0.09,  3.6)
};

float3 DirLatLon(float latDeg, float lonDeg)
{
    float la = radians(latDeg), lo = radians(lonDeg);
    return float3(cos(la) * sin(lo), sin(la), -cos(la) * cos(lo));   // GeoMath.Dir
}

float3 SwirlPoint(float3 p, float weather, out float core)
{
    float t = _TycoonCloudMorph;
    core = 0.0;
    [unroll] for (int i = 0; i < 14; i++)
    {
        float4 v = kVortex[i];
        float hLat = TyHash(float3(i, weather, 3.1)), hLon = TyHash(float3(i, weather, 8.2));
        float hStr = TyHash(float3(i, weather, 5.5)), hOn = TyHash(float3(i, weather, 1.7));
        float3 c = DirLatLon(v.x + (hLat - 0.5) * 24.0, v.y + (hLon - 0.5) * 60.0);
        float cd = dot(p, c);
        if (cd > cos(v.z * 2.5))
        {
            float d = acos(clamp(cd, -1.0, 1.0)) / v.z;
            float spin = v.w * (0.6 + 0.8 * hStr) * step(0.2, hOn) * (0.9 + 0.2 * sin(t * 0.03 + i * 1.7));
            float th = spin * exp(-d * d);
            core = max(core, exp(-d * d * 0.9));   // wolkengefuelltes Zentrum
            float s, co;
            sincos(th, s, co);
            p = normalize(p * co + cross(c, p) * s + c * (cd * (1.0 - co)));   // Rodrigues-Drehung um das Zentrum
        }
    }
    return p;
}

float Fbm(float3 p, float ps, float cyclesPerUnit, int octaves)   // ps = length(fwidth(N)) = Einheitskugel pro Pixel
{
    const float3x3 rot = float3x3(0.00, 0.80, 0.60, -0.80, 0.36, -0.48, -0.60, -0.48, 0.64);   // Oktaven drehen, kein Gitter
    float sum = 0.0, norm = 0.0, amp = 0.5, freq = 1.0;
    [unroll] for (int o = 0; o < 6; o++)
    {
        if (o < octaves)
        {
            float fade = saturate(1.0 - ps * cyclesPerUnit * freq * 0.9);
            sum += amp * fade * TyNoise(p);
            norm += amp * fade;
        }
        p = mul(rot, p) * 2.0;
        amp *= 0.5;
        freq *= 2.0;
    }
    return sum / max(norm, 1e-3);
}

float CloudNoise(float3 Nd, float ps, int octaves, float weather, float lat)
{
    float t = _TycoonCloudMorph;
    float core;
    float3 qs = SwirlPoint(Nd, weather, core);
    float3 q = qs + TyHash(float3(weather, 2.2, 9.9)) * 40.0;        // neue Wetterlage = anderer Rauschbereich
    float3 wv = float3(TyNoise(q * 6.0 + float3(t * 0.020, 0, 0)),
                       TyNoise(q * 6.0 + 17.0 + float3(0, t * 0.016, 0)),
                       TyNoise(q * 6.0 + 41.0 + float3(0, 0, t * 0.024))) - 0.5;
    float syn = Fbm(q * 6.0 + wv * 1.1, ps, 6.0, min(octaves, 3));
    // WICHTIG: den Rauschversatz NICHT vor normalize() addieren - das machte die feinen Wolken zu glatten Flecken
    float3 qf = normalize(lerp(Nd, qs, 0.8));
    float fine = Fbm(qf * 26.0 + wv * 1.6 + TyHash(float3(weather, 4.4, 1.1)) * 40.0, ps, 26.0, octaves - 2);
    float tr = 1.0 - smoothstep(0.35, 0.70, abs(lat));               // Tropen -> klumpig
    float n = lerp(syn * 0.4 + fine * 0.6, syn * 0.25 + fine * 0.75, tr);
    return (n - 0.5) * (1.5 + 0.25 * tr) + 0.5 + core * 0.12;
}

float CloudDensity(float3 N, float lat, float ps, int octaves)
{
    float k = clamp(abs(lat), 0.0, 1.047);
    float wind = -cos(3.0 * k);                                                 // + = ostwaerts
    float merid = -clamp(lat / 0.15, -1.0, 1.0) * 0.3 * cos(3.0 * k);          // + = nordwaerts (weich ueber den Aequator)

    float2 eh = float2(-N.z, N.x);
    float3 E = float3(eh.x, 0.0, eh.y) / max(length(eh), 1e-4);
    float3 Nn = cross(E, N);

    const float cycle = 0.3;                                   // rad Drift pro Ueberblend-Periode
    float tau = _TycoonCloudFlow / cycle;
    float ph0 = frac(tau), ph1 = frac(tau + 0.5);
    float w0 = smoothstep(0.2, 0.8, 1.0 - abs(2.0 * ph0 - 1.0));   // meiste Zeit in einem reinen Wetterzustand
    float d0 = (ph0 - 0.5) * cycle, d1 = (ph1 - 0.5) * cycle;
    float3 N0 = normalize(N - E * (wind * d0) - Nn * (merid * d0));
    float3 N1 = normalize(N - E * (wind * d1) - Nn * (merid * d1));
    float k0 = floor(tau), k1 = floor(tau + 0.5);              // Wetterlage jeder Kopie

    float c = saturate(CloudCoverageLat(lat) + _CloudAmount - 0.5);
    float thr = lerp(0.80, 0.50, c);
    float d_0 = smoothstep(thr - 0.05, thr + 0.32, CloudNoise(N0, ps, octaves, k0, lat));
    float d_1 = smoothstep(thr - 0.05, thr + 0.32, CloudNoise(N1, ps, octaves, k1, lat));
    return lerp(d_1, d_0, w0);
}
